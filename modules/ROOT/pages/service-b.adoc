function basculerContent(caseCochee) {
  const fond = document.getElementById("fond-switch");
  const curseur = document.getElementById("curseur-switch");
  const zoneTexte = document.getElementById("statut-content");
  const formNote = document.getElementById("form-note");
  const conteneurGraphe = document.getElementById("conteneur-graphe");

  if (caseCochee.checked) {
    fond.style.backgroundColor = "#2b7a0b";
    curseur.style.transform = "translateX(24px)";
    zoneTexte.value = "Statut : Oui (Activé - Client content)";
    formNote.style.display = "block"; // Affiche le formulaire de note
  } else {
    fond.style.backgroundColor = "#ccc";
    curseur.style.transform = "translateX(0px)";
    zoneTexte.value = "Statut : Non (Désactivé)";
    formNote.style.display = "none"; // Masque le formulaire si on repasse sur Non
    conteneurGraphe.style.display = "none";
  }
}

function afficherBarreNote(event) {
  event.preventDefault(); // Empêche le rechargement de la page
  
  const note = parseInt(document.getElementById("input-note").value, 10);
  const conteneurGraphe = document.getElementById("conteneur-graphe");
  const barre = document.getElementById("barre-couleur");
  const libelle = document.getElementById("libelle-note");

  // Détermination de la couleur selon le barème
  let couleur = "#000000"; // Noir par défaut (1 à 3)
  let appreciation = "Insuffisant";

  if (note >= 1 && note <= 3) {
    couleur = "#111111"; // Noir
    appreciation = "Faible";
  } else if (note >= 4 && note <= 6) {
    couleur = "#d93025"; // Rouge
    appreciation = "Moyen";
  } else if (note >= 7 && note <= 8) {
    couleur = "#f29900"; // Orange
    appreciation = "Bon";
  } else if (note >= 9 && note <= 10) {
    couleur = "#2b7a0b"; // Vert
    appreciation = "Excellent";
  }

  // Calcul de la largeur en pourcentage (ex: note 7 -> 70%)
  const pourcentage = note * 10;

  conteneurGraphe.style.display = "block";
  libelle.textContent = "Résultat : " + note + " / 10 (" + appreciation + ")";
  barre.style.backgroundColor = couleur;
  barre.style.width = pourcentage + "%";
  barre.textContent = note + "/10";
}
</script>
