<script>
const formC = document.getElementById("formulaire-produit-c");

formC.addEventListener("submit", async function(event) {
  event.preventDefault();
  
  const zoneTexte = document.getElementById("zone-resultat-c");
  const bouton = document.getElementById("btn-submit-c");
  const donnees = new FormData(event.target);

  // Récupération des valeurs avant la réinitialisation du formulaire
  const emailSaisi = donnees.get("email");
  const optionsCoches = [];
  if (donnees.get("option_1")) optionsCoches.push(donnees.get("option_1"));
  if (donnees.get("option_2")) optionsCoches.push(donnees.get("option_2"));
  if (donnees.get("option_3")) optionsCoches.push(donnees.get("option_3"));

  bouton.disabled = true;
  zoneTexte.value = "Envoi des données vers Formspree en cours...";

  try {
    const reponse = await fetch(event.target.action, {
      method: formC.method,
      body: donnees,
      headers: { 'Accept': 'application/json' }
    });

    if (reponse.ok) {
      const listeOptions = optionsCoches.length > 0 ? optionsCoches.join(", ") : "Aucune option";
      zoneTexte.value = "STATUT : Envoyé avec succès\n" +
                        "E-MAIL : " + emailSaisi + "\n" +
                        "OPTIONS : " + listeOptions;
      formC.reset();
    } else {
      zoneTexte.value = "STATUT : Erreur lors de l'envoi vers Formspree.";
    }
  } catch (erreur) {
    zoneTexte.value = "STATUT : Échec de connexion réseau.";
  } finally {
    bouton.disabled = false;
  }
});
</script>
