<script>
const form = document.getElementById("mon-formulaire");

form.addEventListener("submit", async function(event) {
  event.preventDefault(); // Bloque la redirection vers la page Formspree
  
  const statut = document.getElementById("statut-envoi");
  const bouton = document.getElementById("btn-submit");
  const donnees = new FormData(event.target);

  bouton.disabled = true;
  statut.style.color = "#555";
  statut.textContent = "Envoi en cours...";

  try {
    const reponse = await fetch(event.target.action, {
      method: form.method,
      body: donnees,
      headers: { 'Accept': 'application/json' }
    });

    if (reponse.ok) {
      statut.style.color = "#2b7a0b";
      statut.textContent = "Merci ! Votre sélection a bien été envoyée.";
      form.reset(); // Vide les champs et décoche les cases
    } else {
      statut.style.color = "#c0392b";
      statut.textContent = "Erreur lors de l'envoi. Vérifiez votre formulaire.";
    }
  } catch (erreur) {
    statut.style.color = "#c0392b";
    statut.textContent = "Impossible de joindre le serveur. Vérifiez votre connexion.";
  } finally {
    bouton.disabled = false;
  }
});
</script>
